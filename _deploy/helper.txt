

/**
 * 写文件。
 *
 * 这台服务器上 PHP 直接 file_put_contents 到插件目录会 Permission denied
 * （PHP-FPM 的用户和文件属主不是一个），但 WordPress 自己的文件编辑器写得进去 ——
 * 它走的是 WP_Filesystem 这一层。所以这里也走它，直接写只作为兜底。
 */
function yf_gh_write( $dest, $data ) {
	global $wp_filesystem;
	if ( empty( $wp_filesystem ) ) {
		require_once ABSPATH . 'wp-admin/includes/file.php';
		WP_Filesystem();
	}
	if ( ! empty( $wp_filesystem ) && $wp_filesystem->put_contents( $dest, $data, defined( 'FS_CHMOD_FILE' ) ? FS_CHMOD_FILE : 0644 ) ) {
		return true;
	}
	return false !== file_put_contents( $dest, $data ); // phpcs:ignore
}
